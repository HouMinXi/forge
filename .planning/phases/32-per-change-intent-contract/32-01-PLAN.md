    )

    def _merge_contract_spec(yaml_digest, file_content, backend=None, warn_fn=None):
        Merge contracts.yaml digest with --contract file content, optionally
        summarize large file content, and append the bias directive.

        If file_content is non-empty and len(file_content) > 4096 and backend
        is not None: attempt summarization via llm_invoke. On failure, fall
        back to raw file_content and call warn_fn if provided.

        merged = ""
        if yaml_digest:
            merged = yaml_digest
        if file_content:
            # Summarize if large and backend available
            effective_content = file_content
            if len(file_content) > 4096 and backend is not None:
                try:
                    from .llm_invoke import llm_invoke
                    result = llm_invoke(
                        "Summarize the following contract to its key "
                        "invariants and residual risks:\n" + file_content,
                        backend=backend,
                    )
                    effective_content = str(result.content)
                except Exception:
                    if warn_fn:
                        warn_fn("contract: summarization failed, "
                                "injecting raw content")
            merged = (merged + "\n\n" if merged else "") + effective_content
        if merged:
            merged += _CONFIRMATION_BIAS_DIRECTIVE
        return merged
