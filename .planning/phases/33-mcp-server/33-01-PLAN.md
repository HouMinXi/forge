start_job(comm_task: asyncio.Task, proc: asyncio.subprocess.Process,
          tempfile_path: str | None = None) -> str:
  Per D-33-11: receives the SHIELDED comm_task (not just the Process).
  Generate job_id via uuid.uuid4(). Store in _jobs[job_id]:
    {"comm_task": comm_task, "proc": proc, "status": "running",
     "result": None, "created_at": time.monotonic(),
     "tempfile_path": tempfile_path}
  Schedule _wait_for_job(job_id) via asyncio.create_task(). Return job_id.

get_job(job_id: str) -> dict | None:
  Per D-33-15: pop-on-retrieval for terminal states. Run _evict_stale()
  on every call (safe sweep point -- ensures TTL fires even if _wait_for_job
  is hung). If _jobs[job_id] exists and status is "completed" or "failed",
  pop it (remove from dict) and return. If status is "running", return
  without removing. If job_id unknown, return None.

async _wait_for_job(job_id: str) -> None:
  Per D-33-22: await the stored inner_task (the asyncio.Task created by
  create_task(proc.communicate()) before shield), NOT a fresh communicate().
  Per D-33-25: wrap in try/except BaseException. On exception: set status
  "failed", store error message in result, delete tempfile.
  try:
    stdout_bytes, stderr_bytes = await _jobs[job_id]["comm_task"]
    proc = _jobs[job_id]["proc"]
    exit_code = proc.returncode if proc.returncode is not None else -1
    Build result dict with stdout (decoded), stderr (decoded), exit_code,
    verdict via exit_to_verdict(). Set status to "completed".
  except BaseException as exc:
    Set status to "failed", result = {"stdout": "", "stderr": str(exc),
    "exit_code": -1, "verdict": "UNKNOWN(-1)"}.
  finally:
    Per D-33-18: if tempfile_path is set, delete it here (os.unlink, ignore
    FileNotFoundError). The handler does NOT delete tempfiles for timed-out jobs.
  Per D-33-15: also run _evict_stale() after updating.

_evict_stale() -> None:
  Per D-33-15, D-33-32: iterate _jobs, remove entries where
  time.monotonic() - created_at > _JOB_TTL_SECONDS AND status is terminal
  (completed/failed). Running jobs older than TTL: call proc.kill() but
  do NOT remove from _jobs -- let _wait_for_job handle the kill, update
  status to "failed", and a subsequent evict pass garbage-collects the
  terminal entry. This prevents KeyError race with _wait_for_job.

async cleanup_all() -> None:
  Per D-33-14: called from lifespan teardown. Iterate _jobs, for each entry
  with a proc: proc.terminate(), await proc.wait() with 5s timeout, then
  proc.kill() on TimeoutError. Clear _jobs.

  def _check_backend() -> None:

    import cli; cli._load_gate_backends(...)` per D-33-24), CliError from

CLI RUNNER HELPERS:
  async def _run_cli_simple(*args: str) -> tuple[str, str, int]:
    Per D-33-01, D-33-10: asyncio.create_subprocess_exec("code-forge", *args,
    stdout=PIPE, stderr=PIPE). Await proc.communicate(). Return
    (stdout.decode(), stderr.decode(), proc.returncode or 0).

  async def _run_cli_budgeted(*args: str, budget: float = 20.0)
      -> tuple[str, int, float] | tuple[asyncio.Task, asyncio.subprocess.Process]:
    Per D-33-06, D-33-22 (supersedes D-33-11 shield placement):
    proc = await asyncio.create_subprocess_exec("code-forge", *args,
      stdout=PIPE, stderr=PIPE)
    start = time.monotonic()
    # D-33-22: create_task FIRST, then shield wraps it for wait_for.
    # On timeout, inner_task survives (shield protects from cancellation).
    # OLD pattern ensure_future(shield(coro)) left comm_task cancelled.
    inner_task = asyncio.create_task(proc.communicate())
    try:
      stdout_bytes, _ = await asyncio.wait_for(
          asyncio.shield(inner_task), timeout=budget)
      elapsed = time.monotonic() - start
      return (stdout_bytes.decode(), proc.returncode or 0, elapsed)
    except asyncio.TimeoutError:
      # D-33-22: pass inner_task (alive), not the cancelled shield wrapper
      return (inner_task, proc)
    except asyncio.CancelledError:
      # D-33-29: IDE client abort -- kill proc and inner_task, re-raise
      proc.kill()
      inner_task.cancel()
      raise
