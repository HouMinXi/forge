3. HOW TO RUN FORGE  (copy-paste on-ramp -- this is the un-obvious part)
----------------------------------------------------------------
  # 1. work in a linked worktree (review REFUSES to run in the main tree):
  git -C ~/code/forge worktree add .worktrees/<name> -b <branch>
  cd ~/code/forge/.worktrees/<name>

  # 2. give the worktree the trusted gate.yaml (it is gitignored, not checked
  #    out). Symlink ONLY gate.yaml, never the whole dir (nested-symlink trap):
  mkdir -p .code-forge
  ln -s ../../../.code-forge/gate.yaml .code-forge/gate.yaml
  # trust still holds: is_trusted resolves the symlink to the main realpath.

  # 3. export the backend key (cwd-local; env does NOT persist between shells):
  export DEEPSEEK_API_KEY=$(pass show api/deepseek | head -1)     # default
  export MIMO_PRO_API_KEY=$(pass show api/mimo-pro | head -1)     # for --backend mimo-pro

  # 4. run:
  code-forge resolve-outlet                 # expect: subprocess
  code-forge review --whole-file <file> --max-total-rounds 2          # default backend (deepseek)
  code-forge review --backend mimo-pro --whole-file <file>           # mimo-pro (429-prone, item 2.1)
  code-forge review --committed             # review the last commit (dogfood)
  code-forge gate-check                     # deterministic test gate (R1)

  NOTES: command is `code-forge`, never `forge` (name collision). deepseek =
  reliable 3-pass; mimo-pro = rate-limited (2.1). gate.yaml already TRUSTED
  (trust --status = True); no `code-forge trust` needed.
