## Goal

Ship the unified review-subject model: `(source, baseline) -> (findings, dispositions, fixes)` as a typed contract. Resolve `--baseline` and `--head` (git refs / pseudo-refs / stored snapshot / empty) to concrete review content. Compute `source_hash` (STATE-07) so HOLD resume can detect stale state. Persist + invalidate non-git snapshots (BASELINE-02/03) so v1 ad-hoc script-review workflow survives in v2.

## Requirements (4)

- **BASELINE-01**: Unified review model -- `forge reviews (source, baseline) -> (findings, dispositions, fixes)`. Source = current state of reviewed code. Baseline = git ref / stored snapshot / empty. Git diff is one baseline source; snapshot and "empty" are equally valid. Generalizes v1 (implicitly `baseline=empty`) and v2 git-diff under one model.
- **BASELINE-02**: Non-git mode snapshot persistence -- on PASS, store `(file paths, file content hashes, finding fingerprints + final dispositions)` at `.forge/snapshots/<source-hash>.json`. Subsequent runs auto-detect; `--baseline empty` overrides + forces full re-review.
- **BASELINE-03**: Snapshot invalidation -- snapshot is invalidated and treated as `empty` baseline if any file in snapshot's path list is missing OR any file's content hash differs in a way per-file diff cannot be reconstructed. Partial invalidation (one file changed, others unchanged) is acceptable -- forge uses snapshot for unchanged files, full-review for changed.
- **STATE-07**: state.json is flushed before HOLD; `source_hash` = SHA256 of the review subject (whitespace normalized: trailing-ws strip + LF endings). Git mode: covers `git diff <baseline_resolved>..<head_resolved>` + untracked files via `git diff --no-index /dev/null <untracked>` when `--head=WORKING`. Non-git mode: SHA256 of concatenated file contents (sorted by path) for all `path...` args. Next forge invocation auto-detects `.forge/state.json` (preferred) or cwd `./state.json` and validates stored `source_hash`; match -> resume HOLD with prior dispositions; mismatch -> stale-state warning + start fresh (no `--force` flag in v2.0). Edits during HOLD invalidate resume by design.

## Module Designs

### `src/forge/baseline.py`

```python
"""BaselineSpec discriminated union + resolver.

Owned by 02-03. 02-05 CLI parser emits BaselineSpec from string args; this
module dispatches to the right resolution path.
"""
from __future__ import annotations
from dataclasses import asdict, dataclass
from pathlib import Path
from typing import Optional, Union

from .errors import BaselineResolutionError
from .git import (
    INDEX,
    WORKING,
    cached_diff,
    git_diff,
    is_git_repo,
    is_pseudo_ref,
    resolve_git_ref,
    working_tree_diff,
)
from .snapshot import load_snapshot


@dataclass(frozen=True)
class GitRefBaseline:
    """Baseline = git ref. Accepted: HEAD, branch name, commit sha, WORKING, INDEX."""
    ref: str


@dataclass(frozen=True)
class SnapshotBaseline:
    """Baseline = stored .forge/snapshots/<source-hash>.json."""
    path: Path


@dataclass(frozen=True)
class EmptyBaseline:
    """Baseline = nothing. All current source content is new (BASELINE-01)."""
    pass


BaselineSpec = Union[GitRefBaseline, SnapshotBaseline, EmptyBaseline]


@dataclass(frozen=True)
class ResolvedReview:
    """Concrete review subject + baseline content after resolution.

    Consumed by state machine (02-02) and source_hash (STATE-07).
    """
    source_files: list[Path]               # files in scope after `paths` filter
    baseline_content: Optional[dict]       # None for empty/git baseline; snapshot dict for snapshot baseline
    git_diff: Optional[str]                # unified diff for git mode; None for non-git
    mode_hint: str                         # "git" | "non-git"


def resolve_baseline(
    baseline_spec: BaselineSpec,
    head_spec: Optional["GitRefBaseline"],
    paths: list[Path],
    cwd: Path,
) -> ResolvedReview:
    """Resolve (baseline, head, paths) to concrete review subject.

    Raises:
      BaselineResolutionError: invalid combination (e.g., snapshot baseline + git head).
    """
    if isinstance(baseline_spec, GitRefBaseline):
        return _resolve_git(baseline_spec, head_spec, paths, cwd)
    if isinstance(baseline_spec, SnapshotBaseline):
        if head_spec is not None:
            raise BaselineResolutionError(
                f"SnapshotBaseline does not accept head_spec (got: {head_spec!r}); "
                f"snapshot is its own implicit head"
            )
        return _resolve_snapshot(baseline_spec, paths, cwd)
    if isinstance(baseline_spec, EmptyBaseline):
        if head_spec is not None and not is_git_repo(cwd):
            raise BaselineResolutionError(
                f"EmptyBaseline + head_spec is only valid in a git repo; cwd={cwd}"
            )
        return _resolve_empty(head_spec, paths, cwd)
    raise BaselineResolutionError(f"unknown baseline spec type: {type(baseline_spec)}")


def _resolve_git(
    baseline_spec: GitRefBaseline,
    head_spec: Optional[GitRefBaseline],
    paths: list[Path],
    cwd: Path,
) -> ResolvedReview:
    """B3 resolution -- git mode.

    Algorithm:
      1. Reject if cwd is not in a git repo.
      2. Reject pseudo-ref as baseline (WORKING/INDEX are head-only concepts).
      3. Validate baseline ref via resolve_git_ref (raises if unknown).
      4. Default head = GitRefBaseline(WORKING) if head_spec is None.
      5. Dispatch by head ref kind:
         - WORKING: working_tree_diff(baseline_ref, paths, cwd)  -- tracked + non-binary untracked
         - INDEX:   cached_diff(baseline_ref, paths, cwd)
         - regular: validate head via resolve_git_ref, then git_diff(baseline, head, paths, cwd)
      6. Return ResolvedReview with diff in git_diff field, mode_hint="git".
    """
    if not is_git_repo(cwd):
        raise BaselineResolutionError(
            f"GitRefBaseline used outside git repo (cwd={cwd})"
        )
    if is_pseudo_ref(baseline_spec.ref):
        raise BaselineResolutionError(
            f"baseline cannot be a pseudo-ref ({baseline_spec.ref}); "
            f"pseudo-refs are head-only"
        )
    resolve_git_ref(baseline_spec.ref, cwd)  # raises if ref unknown

    head = head_spec if head_spec is not None else GitRefBaseline(WORKING)
    if head.ref == WORKING:
        diff = working_tree_diff(baseline_spec.ref, paths, cwd)
    elif head.ref == INDEX:
        diff = cached_diff(baseline_spec.ref, paths, cwd)
    else:
        resolve_git_ref(head.ref, cwd)  # raises if ref unknown
        diff = git_diff(baseline_spec.ref, head.ref, paths, cwd)

    return ResolvedReview(
        source_files=paths,
        baseline_content=None,
        git_diff=diff,
        mode_hint="git",
    )


def _resolve_snapshot(
    baseline_spec: SnapshotBaseline,
    paths: list[Path],
    cwd: Path,
) -> ResolvedReview:
    """B3 resolution -- stored snapshot.

    Algorithm:
      1. load_snapshot(path); returns None on missing file (BASELINE-03).
      2. If None: fall back to EmptyBaseline resolution silently (BASELINE-03: absence == empty).
      3. Else: pack snapshot dict into baseline_content for state machine consumption.
    """
    snap = load_snapshot(baseline_spec.path)
    if snap is None:
        # BASELINE-03: snapshot file absence is normal, not an error
        return _resolve_empty(None, paths, cwd)
    return ResolvedReview(
        source_files=paths,
        baseline_content={"snapshot": asdict(snap)},
        git_diff=None,
        mode_hint="non-git",
    )


def _resolve_empty(
    head_spec: Optional[GitRefBaseline],
    paths: list[Path],
    cwd: Path,
) -> ResolvedReview:
    """B3 resolution -- empty baseline (all source is new, per BASELINE-01).

    Algorithm:
      1. mode_hint = "git" if cwd is in a git repo else "non-git".
      2. If git repo AND head_spec provided: still produce diff from empty tree to head
         (use git's empty tree sha 4b825dc642cb6eb9a060e54bf8d69288fbee4904 as baseline).
         This enables --baseline empty --head WORKING for full re-review in a git repo.
      3. Otherwise: no diff (review reads file content directly).
    """
    if is_git_repo(cwd) and head_spec is not None:
        EMPTY_TREE_SHA = "4b825dc642cb6eb9a060e54bf8d69288fbee4904"
        if head_spec.ref == WORKING:
            diff = working_tree_diff(EMPTY_TREE_SHA, paths, cwd)
        elif head_spec.ref == INDEX:
            diff = cached_diff(EMPTY_TREE_SHA, paths, cwd)
        else:
            resolve_git_ref(head_spec.ref, cwd)
            diff = git_diff(EMPTY_TREE_SHA, head_spec.ref, paths, cwd)
        return ResolvedReview(
            source_files=paths,
            baseline_content=None,
            git_diff=diff,
            mode_hint="git",
        )
    mode = "git" if is_git_repo(cwd) else "non-git"
    return ResolvedReview(
        source_files=paths,
        baseline_content=None,
        git_diff=None,
        mode_hint=mode,
    )


def serialize_baseline_spec(spec: BaselineSpec) -> str:
    """OQ1 fix: produce single-line repr for state.json baseline_spec_repr field.

    Formats:
      GitRefBaseline(ref="HEAD")        -> "git:HEAD"
      SnapshotBaseline(path=...)        -> "snapshot:.forge/snapshots/abc.json"  (posix)
      EmptyBaseline()                   -> "empty"

    Consumed by 02-04 (state.py extension stores this string in state.json so HOLD
    resume can record which baseline was used).
    """
    if isinstance(spec, GitRefBaseline):
        return f"git:{spec.ref}"
    if isinstance(spec, SnapshotBaseline):
        return f"snapshot:{spec.path.as_posix()}"
    if isinstance(spec, EmptyBaseline):
        return "empty"
    raise BaselineResolutionError(f"cannot serialize unknown spec type: {type(spec)}")
```

### `src/forge/source.py`

```python
"""source_hash computation per STATE-07.

Whitespace normalization: trailing-ws strip + LF line endings.
H1/H3 fixes applied: binary files hashed as raw bytes (preserves invalidation
correctness); path serialization uses as_posix() for cross-platform determinism.
"""
from __future__ import annotations
import hashlib
from pathlib import Path
from typing import Optional


def normalize_text(text: str) -> str:
    """Strip trailing whitespace per line; force LF endings; no trailing blank line stripping."""
    lines = [line.rstrip() for line in text.splitlines()]
    return "\n".join(lines)


def compute_source_hash(
    *,
    git_diff: Optional[str] = None,
    files: Optional[list[Path]] = None,
) -> str:
    """STATE-07 source_hash.

    Git mode: caller passes git_diff (unified diff output).
    Non-git mode: caller passes files. Files are sorted by posix path string
    for cross-platform deterministic ordering (H3). Binary files (UnicodeDecodeError
    on utf-8 read) are hashed as raw bytes with a binary marker (H1) -- this preserves
    invalidation correctness for binary edits and keeps source_hash stable.

    Exactly one of git_diff / files must be provided. Returns lowercase hex SHA256.
    """
    if (git_diff is None) == (files is None):
        raise ValueError("compute_source_hash: pass exactly one of git_diff or files")

    h = hashlib.sha256()
    if git_diff is not None:
        h.update(b"mode=git\n")
        h.update(normalize_text(git_diff).encode("utf-8"))
        return h.hexdigest()

    h.update(b"mode=non-git\n")
    for f in sorted(files, key=lambda p: p.as_posix()):  # H3: deterministic ordering
        try:
            content = f.read_text(encoding="utf-8")
            h.update(f"--- {f.as_posix()} text\n".encode("utf-8"))
            h.update(normalize_text(content).encode("utf-8"))
        except UnicodeDecodeError:
            # H1: binary file -- hash raw bytes (preserves invalidation correctness)
            h.update(f"--- {f.as_posix()} binary\n".encode("utf-8"))
            h.update(f.read_bytes())
        h.update(b"\n")
    return h.hexdigest()
```

### `src/forge/snapshot.py`

```python
"""Snapshot persistence per BASELINE-02 + invalidation per BASELINE-03.

SCHEMA_VERSION is independent of state.SCHEMA_VERSION (snapshot evolves
on different cadence from state.json).

B2 fix: NO Disposition import. `finding_dispositions: dict[str, str]` stores
disposition values as strings; state machine (02-02) converts to/from Disposition
at the read/write boundary. Pure-str at storage layer simplifies serialization
and avoids cross-sub-plan import coupling.
"""
from __future__ import annotations
from dataclasses import dataclass, field, asdict
from pathlib import Path
from typing import Optional
import hashlib
import json

from .errors import (
    BaselineResolutionError,
    CorruptedSnapshotError,
    SnapshotSchemaMismatchError,
)

SNAPSHOT_SCHEMA_VERSION: int = 1


@dataclass(frozen=True)
class SnapshotEntry:
    """One file's recorded state in the snapshot."""
    path: str                  # path-as-posix string relative to snapshot root (H3 deterministic)
    content_hash: str          # SHA256 (text: normalized; binary: raw bytes -- matches _hash_file)


@dataclass
class Snapshot:
    """A persisted baseline snapshot. BASELINE-02 + BASELINE-03."""
    schema_version: int = SNAPSHOT_SCHEMA_VERSION
    source_hash: str = ""                                    # references the source_hash this snapshot represents
    files: list[SnapshotEntry] = field(default_factory=list)
    finding_dispositions: dict[str, str] = field(default_factory=dict)  # fingerprint -> Disposition.value (str)


@dataclass(frozen=True)
class InvalidationResult:
    """BASELINE-03 partial-invalidation result.

    `missing` = files in snapshot but absent from current source.
    `changed` = files present in both but content hash differs.
    `unchanged` = files where snapshot hash matches current.
    `added` = files in current but not in snapshot.

    Caller (state machine) uses these lists to decide per-file scope.
    """
    missing: list[str]
    changed: list[str]
    unchanged: list[str]
    added: list[str]


def snapshot_path_for(source_hash: str, cwd: Path) -> Path:
    """Standard location: .forge/snapshots/<source-hash>.json under cwd."""
    return cwd / ".forge" / "snapshots" / f"{source_hash}.json"


def find_existing_snapshot(source_hash: str, cwd: Path) -> Optional[Path]:
    """H5 fix: snapshot auto-discovery helper.

    Returns the snapshot path if a snapshot exists for the given source_hash, else None.
    Caller (02-04 / 02-05 orchestration) uses this to decide whether to dispatch
    SnapshotBaseline or EmptyBaseline.
    """
    p = snapshot_path_for(source_hash, cwd)
    return p if p.exists() else None


SNAPSHOT_COUNT_WARN_THRESHOLD: int = 50  # OQ2: warn if snapshot dir holds more than this


def save_snapshot(snapshot: Snapshot, path: Path) -> None:
    """Atomic write of snapshot file. Auto-creates parent dirs (0755 per D1).

    OQ2 fix: after write, count snapshot files in directory; if > threshold,
    warn user about manual cleanup (v2.0 has no auto-GC).
    """
    import warnings
    path.parent.mkdir(parents=True, exist_ok=True, mode=0o755)
    tmp = path.with_suffix(".tmp")
    tmp.write_text(json.dumps(asdict(snapshot), indent=2))
    tmp.replace(path)

    # OQ2: snapshot retention warning (no auto-GC in v2.0)
    snapshots = list(path.parent.glob("*.json"))
    if len(snapshots) > SNAPSHOT_COUNT_WARN_THRESHOLD:
        warnings.warn(
            f"forge: {len(snapshots)} snapshot files in {path.parent}; "
            f"consider manual cleanup (no auto-GC in v2.0)",
            stacklevel=2,
        )


def load_snapshot(path: Path) -> Optional[Snapshot]:
    """Load snapshot. Returns None on missing file (BASELINE-03: absence == empty baseline).

    Raises:
      CorruptedSnapshotError: JSON parse failure.
      SnapshotSchemaMismatchError: schema_version != SNAPSHOT_SCHEMA_VERSION.
    """
    if not path.exists():
        return None
    try:
        data = json.loads(path.read_text())
    except json.JSONDecodeError as e:
        raise CorruptedSnapshotError(f"cannot parse {path}: {e}") from e

    sv = data.get("schema_version")
    if sv != SNAPSHOT_SCHEMA_VERSION:
        raise SnapshotSchemaMismatchError(
            f"snapshot schema_version={sv}, forge expects {SNAPSHOT_SCHEMA_VERSION}; "
            f"remove {path} to start fresh"
        )

    return Snapshot(
        schema_version=data["schema_version"],
        source_hash=data["source_hash"],
        files=[SnapshotEntry(**e) for e in data.get("files", [])],
        finding_dispositions=dict(data.get("finding_dispositions", {})),
    )


def validate_snapshot(
    snapshot: Snapshot, current_files: list[Path], root: Path
) -> InvalidationResult:
    """BASELINE-03: classify each current file as unchanged / changed / added.
    Snapshot files absent from current are `missing`.

    H6 fix: files outside `root` raise BaselineResolutionError with explicit context
    (previously raised ValueError from relative_to, leaking implementation detail).
    """
    snapshot_map = {e.path: e.content_hash for e in snapshot.files}
    current_map: dict[str, str] = {}
    for f in current_files:
        try:
            rel = f.relative_to(root).as_posix()  # H3: posix-form path key
        except ValueError as e:
            raise BaselineResolutionError(
                f"file {f} is outside snapshot root {root} -- cannot classify against snapshot"
            ) from e
        current_map[rel] = _hash_file(f)

    missing = sorted(p for p in snapshot_map if p not in current_map)
    added = sorted(p for p in current_map if p not in snapshot_map)
    changed = sorted(
        p for p in current_map if p in snapshot_map and current_map[p] != snapshot_map[p]
    )
    unchanged = sorted(
        p for p in current_map if p in snapshot_map and current_map[p] == snapshot_map[p]
    )

    return InvalidationResult(
        missing=missing, changed=changed, unchanged=unchanged, added=added
    )


def _hash_file(path: Path) -> str:
    """SHA256 of file content.

    Text files use compute_source_hash normalization (LF + trailing-ws strip).
    Binary files hash raw bytes (H1 fix -- preserves invalidation correctness for
    binary edits; matches compute_source_hash binary handling).
    """
    from .source import normalize_text  # local import to avoid cycle if module split
    try:
        content = path.read_text(encoding="utf-8")
        return hashlib.sha256(normalize_text(content).encode("utf-8")).hexdigest()
    except UnicodeDecodeError:
        # H1: binary file -- hash raw bytes for invalidation correctness
        return hashlib.sha256(path.read_bytes()).hexdigest()
```

```python
"""Phase 1 module. 02-03 adds repo detection, ref validation, pseudo-ref resolution.

Existing API unchanged. New surface (B1 + H2 fixes):
  - is_git_repo(cwd)             -> bool                   (B1)
  - resolve_git_ref(ref, cwd)    -> str (resolved sha)     (B1; raises on unknown ref)
  - is_pseudo_ref(name)          -> bool
  - working_tree_diff(...)       -> str                    (H2: binary untracked skipped + warned)
  - cached_diff(...)             -> str
  - git_diff(baseline, head, ...) -> str                   (helper for non-pseudo head)
"""
# ... existing Phase 1 code ...
import subprocess
import warnings
from pathlib import Path


WORKING = "WORKING"
INDEX = "INDEX"
PSEUDO_REFS = {WORKING, INDEX}


def is_pseudo_ref(name: str) -> bool:
    return name in PSEUDO_REFS


def is_git_repo(cwd: Path) -> bool:
    """B1 fix: check whether cwd is inside a git repo.

    Implementation: `git rev-parse --git-dir` returns non-zero outside a repo.
    """
    try:
        result = subprocess.run(
            ["git", "rev-parse", "--git-dir"],
            cwd=cwd,
            capture_output=True,
            text=True,
            check=False,
        )
        return result.returncode == 0
    except (FileNotFoundError, OSError):
        # git not installed or cwd inaccessible
        return False


def resolve_git_ref(ref: str, cwd: Path) -> str:
    """B1 fix: validate that a git ref exists; return its resolved sha.

    Raises:
      BaselineResolutionError: ref does not exist (delegated import to avoid cycle).
    """
    from .errors import BaselineResolutionError  # local import: errors.py imports nothing from git
    result = subprocess.run(
        ["git", "rev-parse", "--verify", ref + "^{commit}"],
        cwd=cwd,
        capture_output=True,
        text=True,
        check=False,
    )
    if result.returncode != 0:
        raise BaselineResolutionError(
            f"git ref {ref!r} does not resolve in {cwd}: {result.stderr.strip()}"
        )
    return result.stdout.strip()


def _is_likely_binary(path: Path) -> bool:
    """H2 fix: heuristic binary detection via null-byte presence in first 8KB.

    Matches git's own diff-detection behavior (loosely). Used to skip binary
    untracked files in working_tree_diff to avoid garbage diff output.
    """
    try:
        with open(path, "rb") as f:
            chunk = f.read(8192)
        return b"\0" in chunk
    except OSError:
        return False  # unreadable -- treat as text (downstream tools handle)


def git_diff(baseline_ref: str, head_ref: str, paths: list[Path], repo_root: Path) -> str:
    """Standard `git diff <baseline_ref> <head_ref> -- <paths>` invocation.

    Exit code semantics (R3-1 fix; matches Phase 1 run_git_diff per Mimo F-03 in
    src/forge/git.py:80-86):
      0 = no differences (return empty stdout)
      1 = differences found (NORMAL -- return stdout with diff)
      2+ = real git error (raise BaselineResolutionError)
    """
    from .errors import BaselineResolutionError
    cmd = ["git", "diff", baseline_ref, head_ref, "--"] + [str(p) for p in paths]
    result = subprocess.run(cmd, cwd=repo_root, capture_output=True, text=True, check=False)
    if result.returncode not in (0, 1):
        raise BaselineResolutionError(
            f"git diff {baseline_ref}..{head_ref} failed "
            f"(exit {result.returncode}): {result.stderr.strip()}"
        )
    return result.stdout


def cached_diff(baseline_ref: str, paths: list[Path], repo_root: Path) -> str:
    """`git diff --cached <baseline_ref> -- <paths>` (staged changes vs baseline).

    Exit code semantics (R3-1 fix): accept 0/1, raise on 2+.
    """
    from .errors import BaselineResolutionError
    cmd = ["git", "diff", "--cached", baseline_ref, "--"] + [str(p) for p in paths]
    result = subprocess.run(cmd, cwd=repo_root, capture_output=True, text=True, check=False)
    if result.returncode not in (0, 1):
        raise BaselineResolutionError(
            f"git diff --cached {baseline_ref} failed "
            f"(exit {result.returncode}): {result.stderr.strip()}"
        )
    return result.stdout


def working_tree_diff(baseline_ref: str, paths: list[Path], repo_root: Path) -> str:
    """Diff baseline..working_tree, including non-binary untracked files.

    Tracked diff: `git diff <baseline_ref> -- <paths>` (git emits "Binary files differ"
                  for tracked binaries naturally; no extra filtering needed).
    Untracked:    enumerate via `git ls-files --others --exclude-standard -- <paths>`,
                  synthesize as full-add via `git diff --no-index /dev/null <file>`.
                  H2 fix: binary untracked files are SKIPPED (with warnings.warn)
                  to avoid `git diff --no-index` producing garbage on binary content.
    Concatenate. Output is stable across invocations for same input.
    """
    # R3-1 fix: tracked diff has same exit-code semantics as standalone git_diff
    # (exit 1 = diff exists is NORMAL, not an error). Must NOT use check=True.
    from .errors import BaselineResolutionError
    tracked_cmd = ["git", "diff", baseline_ref, "--"] + [str(p) for p in paths]
    tracked_result = subprocess.run(
        tracked_cmd, cwd=repo_root, capture_output=True, text=True, check=False
    )
    if tracked_result.returncode not in (0, 1):
        raise BaselineResolutionError(
            f"git diff {baseline_ref} (tracked, working_tree_diff) failed "
            f"(exit {tracked_result.returncode}): {tracked_result.stderr.strip()}"
        )
    tracked = tracked_result.stdout

    ls_cmd = ["git", "ls-files", "--others", "--exclude-standard", "--"] + [str(p) for p in paths]
    untracked_paths = [
        line
        for line in subprocess.run(
            ls_cmd, cwd=repo_root, capture_output=True, text=True, check=True
        ).stdout.splitlines()
        if line.strip()
    ]

    untracked_diffs: list[str] = []
    skipped_binary: list[str] = []
    for rel_path in sorted(untracked_paths):  # sort for determinism (H3-adjacent)
        full = repo_root / rel_path
        if _is_likely_binary(full):
            skipped_binary.append(rel_path)
            continue
        # `git diff --no-index` exit codes (R2 DeepSeek fix):
        #   0 = files identical (impossible vs /dev/null when file has content)
        #   1 = files differ (THE expected case for new untracked text -> emit add hunks)
        #   2+ = real error (file unreadable, git internal failure, etc.)
        # check=True would raise on the normal exit-1 path; blind check=False would
        # swallow real exit-2+ errors. Must accept 0/1 and explicitly raise on 2+.
        cmd = ["git", "diff", "--no-index", "/dev/null", str(full)]
        result = subprocess.run(
            cmd, cwd=repo_root, capture_output=True, text=True, check=False
        )
        if result.returncode not in (0, 1):
            from .errors import BaselineResolutionError
            raise BaselineResolutionError(
                f"git diff --no-index failed for untracked file {rel_path} "
                f"(exit {result.returncode}): {result.stderr.strip()}"
            )
        untracked_diffs.append(result.stdout)

    if skipped_binary:
        warnings.warn(
            f"forge: skipped {len(skipped_binary)} binary untracked file(s) from working-tree diff: "
            f"{skipped_binary[:3]}{'...' if len(skipped_binary) > 3 else ''}",
            stacklevel=2,
        )

    return tracked + "\n".join(untracked_diffs)
```
