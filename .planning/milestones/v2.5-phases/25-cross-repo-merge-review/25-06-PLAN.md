```python
def format_cross_repo_output(
    per_repo_findings: dict,  # {label: [finding_dict, ...]} from per_cwd receipts
    ordered_labels: list,
    output_fn=print,
) -> None:
    """Emit grouped verdict output per D-12.
    For each label in ordered_labels (primary first, then siblings):
        output_fn("=== [%s] ===" % label)
        for finding in per_repo_findings.get(label, []):
            # Format: [label] file:line -- description (D-13 attribution preserved)
            file_ref = finding.get("file", "?")
            line_ref = finding.get("line", 0)
            desc = finding.get("description", "")
            # format_cross_repo_output() ADDS [label] prefix deterministically.
            # D-08 section headers convey label context to the LLM, but do NOT
            # mechanically inject [label] into file paths in the diff text.
            # Receipt finding["file"] is a plain path (e.g. "src/a.py").
            # Attribution is achieved here, not by receipt data.
            output_fn("[%s] %s:%s -- %s" % (label, file_ref, line_ref, desc))
    Missing label -> header emitted with no body (graceful).
    """
```
