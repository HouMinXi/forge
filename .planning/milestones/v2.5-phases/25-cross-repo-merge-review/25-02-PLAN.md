```python
def get_sibling_diff(repo_path: Path, ref_spec: str) -> str:
    """Acquire unified diff for a sibling repo.
    repo_path: resolved absolute path to sibling repo dir.
    ref_spec: "baseline..head" format (e.g. "main..feature-x").
    Returns unified diff string (empty string if no changes).
    Raises ValueError if ref_spec format is wrong.
    Raises BaselineResolutionError (from git.py) if ref does not exist.
    Never catches BaselineResolutionError -- caller sees it directly (D-06 fail-closed)."""

def build_cross_repo_context(
    repos: list[dict],
) -> str:
    """Assemble joint review context string from per-repo diffs.
    repos: list of {"label": str, "ref": str, "diff": str} dicts.
    Returns a string with:
      - Summary header: "Cross-repo review: label1 (ref1) + label2 (ref2)"
      - Per-repo diff stats line: "label: N files changed, +X/-Y (file1, file2, ...)"
      - Labeled diff blocks: "## Repo: [label] (ref)" followed by the diff
    File paths inside each diff block are NOT prefixed in the diff text itself;
    the [label] prefix is conveyed by the section heading (D-07/D-08).
    Empty diff for a repo still emits its heading ("## Repo: [label] (ref)\n(no changes)")."""

def make_per_repo_cwd(label: str, gate_config: dict | None = None) -> Path:
    """Create isolated .code-forge/ work dir for a single StateMachine thread.
    Returns a temp dir path with .code-forge/ subdir pre-created.
    Caller must clean up via tempfile.TemporaryDirectory or shutil.rmtree.
    Uses tempfile.mkdtemp(prefix="forge-cross-{label}-")."""
```
